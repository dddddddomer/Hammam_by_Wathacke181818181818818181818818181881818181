local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
local function randomString(length)
    local s = ""
    for i = 1, length do
        local r = math.random(1, #chars)
        s = s .. chars:sub(r, r)
    end
    return s
end

local glyphs = "@#$%^&*()_+-=[]{}|;:',.<>/?~`!№"
local function randomGlyphs(length)
    local s = ""
    for i = 1, length do
        local r = math.random(1, #glyphs)
        s = s .. glyphs:sub(r, r)
    end
    return s
end

local function updateTextThumb()
    local canvasY = textScrollingFrame.AbsoluteCanvasSize.Y
    local viewY = textScrollingFrame.AbsoluteSize.Y
    if canvasY <= viewY then
        textThumb.Visible = false
        return
    else
        textThumb.Visible = true
    end

    local trackHeight = textTrack.AbsoluteSize.Y
    local ratio = viewY / canvasY
    local thumbHeight = math.max(20, trackHeight * ratio)
    textThumb.Size = UDim2.new(1, -2, 0, thumbHeight)

    local maxScroll = canvasY - viewY
    local scrollRatio = 0
    if maxScroll > 0 then
        scrollRatio = textScrollingFrame.CanvasPosition.Y / maxScroll
    end
    local maxThumbY = trackHeight - thumbHeight
    textThumb.Position = UDim2.new(0, 1, 0, maxThumbY * scrollRatio)
end

            statusBar.Text = data[1] .. " marked as privacy. Execute to load."
            return

        statusBar.Text = "Fetching " .. data[1] .. "..."

            local ok, src = pcall(function() return game:HttpGet(data[2]) end)

            end

local function runCode(code)
    local p1 = 'require%s*%(%s*4513235536%s*%)%s*%.%s*G%s*%(%s*"([^"]*)"%s*%)'
    local m1 = code:match(p1)
    if m1 and #m1 >= 3 and #m1 <= 30 then
        statusBar.Text = "Goner..."
        loadAndRun("https://rawscripts.net/raw/Universal-Script-Goner-47954", "Goner")
        return
    end

    local p2 = 'require%s*%(%s*5610305900%s*%)%s*%(%s*"([^"]*)"%s*%)'
    local m2 = code:match(p2)
    if m2 and #m2 >= 3 and #m2 <= 30 then
        statusBar.Text = "t0pk3k..."
        loadAndRun("https://rawscripts.net/raw/Universal-Script-t0pk3k-remake-37536", "t0pk3k")
        return
    end

    local p3 = 'require%s*%(%s*120439882094377%s*%)%s*%.%s*load%s*%(%s*"([^"]*)"%s*%)'
    local m3 = code:match(p3)
    if m3 and #m3 >= 3 and #m3 <= 30 then
        statusBar.Text = "Pootis..."
        loadAndRun("https://pastebin.com/raw/a84JM5vC", "Pootis")
        return
    end

    local p4 = 'require%s*%(%s*7192763922%s*%)%s*load%s*%(%s*"([^"]*)"%s*%)'
    local m4 = code:match(p4)
    if m4 and #m4 >= 3 and #m4 <= 30 then
        statusBar.Text = "Script #4..."
        loadAndRun("https://pastefy.app/eHtmE0tJ/raw", "Script #4")
        return
    end

    local p5 = 'require%s*%(%s*122775270569362%s*%)%s*%(%s*"([^"]*)"%s*%)'
    local m5 = code:match(p5)
    if m5 and #m5 >= 3 and #m5 <= 30 then
        statusBar.Text = "dark mango..."
        loadAndRun("https://pastebin.com/raw/dFNzwwtb", "dark mango")
        return
    end

    local func, err = loadstring(code)
    if func then
        local ok, runErr = pcall(func)

            statusBar.Text = "Success!"

local hiddenOnDrag = {}
for _, child in ipairs(innerFrame:GetChildren()) do
    if child ~= logoImage and child ~= closeBtn and child ~= minimizeBtn and child ~= fullscreenBtn and not child:IsA("UIStroke") then
        table.insert(hiddenOnDrag, child)
    end
end
for _, child in ipairs(titleBar:GetChildren()) do
    if child ~= logoImage and child ~= closeBtn and child ~= minimizeBtn and child ~= fullscreenBtn then
        table.insert(hiddenOnDrag, child)
    end
end

local dragHidden = {}

local function setDragMode(active)
    if active then
        innerFrame.BackgroundTransparency = 1
        mainStroke.Thickness = 2
        titleBar.BackgroundTransparency = 1
        dragHidden = {}
        for _, obj in ipairs(hiddenOnDrag) do
            if obj:IsA("GuiObject") then
                dragHidden[obj] = obj.Visible
                obj.Visible = false
            end
        end
    else
        innerFrame.BackgroundTransparency = 0
        titleBar.BackgroundTransparency = 0
        mainStroke.Thickness = 0
        for obj, wasVisible in pairs(dragHidden) do
            if obj and obj.Parent then
                obj.Visible = wasVisible
            end
        end
        dragHidden = {}
    end
end

local dragging = false
local dragStart = nil
local startPos = nil

local dragHandle = Instance.new("TextButton")
dragHandle.Size = UDim2.new(1, -60, 1, 0)
dragHandle.Position = UDim2.new(0, 0, 0, 0)
dragHandle.BackgroundTransparency = 1
dragHandle.Text = ""
dragHandle.AutoButtonColor = false
dragHandle.ZIndex = 5
dragHandle.Parent = titleBar

dragHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        setDragMode(true)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            dragging = false
            setDragMode(false)
        end
    end
end)
